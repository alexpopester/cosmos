# Telemetry header: STREAM_ID, SEQUENCE, PKT_LEN (big endian), cFE time (4 + 2 bytes), 4 spare: payload at byte 16.
# cFE time here is mission elapsed time since the flight software started, not UTC: COSMOS uses the received time.
<%
  def cfs_tlm_header(mid)
    [
      "  APPEND_ID_ITEM STREAM_ID 16 UINT #{mid} \"Message id\" BIG_ENDIAN",
      "  APPEND_ITEM SEQUENCE 16 UINT \"CCSDS sequence flags and count\" BIG_ENDIAN",
      "  APPEND_ITEM PKT_LEN 16 UINT \"CCSDS length (total - 7)\" BIG_ENDIAN",
      "  APPEND_ITEM SECONDS 32 UINT \"cFE time: seconds (mission elapsed)\" BIG_ENDIAN",
      "  APPEND_ITEM SUBSECS 16 UINT \"cFE time: subseconds\" BIG_ENDIAN",
      "  APPEND_ITEM SPARE_HDR 32 UINT \"Header padding\"",
    ].join("\n")
  end
%>
