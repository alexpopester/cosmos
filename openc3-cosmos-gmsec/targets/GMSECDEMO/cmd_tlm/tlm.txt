# Plain CCSDS space packets, nothing GMSEC specific. The interface hands COSMOS
# the raw bytes from each message's DATA field and these definitions decode
# them. HEALTH and POWER are told apart by APID (an ID item); EVENT is matched
# by the subject it arrives on instead (META GMSEC_SUBJECT).

TELEMETRY GMSECDEMO HEALTH BIG_ENDIAN "Spacecraft health, identified by APID"
  APPEND_ITEM    CCSDSVER     3 UINT "CCSDS version"
  APPEND_ITEM    CCSDSTYPE    1 UINT "CCSDS packet type"
  APPEND_ITEM    CCSDSSHF     1 UINT "CCSDS secondary header flag"
  APPEND_ID_ITEM CCSDSAPID   11 UINT 100 "CCSDS application process id"
  APPEND_ITEM    CCSDSSEQFLAGS 2 UINT "CCSDS sequence flags"
  APPEND_ITEM    CCSDSSEQCNT 14 UINT "CCSDS packet sequence count"
  APPEND_ITEM    CCSDSLENGTH 16 UINT "CCSDS packet data length"
  APPEND_ITEM    TEMP1       16 INT "Temperature 1"
    POLY_READ_CONVERSION 0.0 0.01
    UNITS CELSIUS C
    LIMITS DEFAULT 1 ENABLED -40.0 -30.0 50.0 60.0
  APPEND_ITEM    TEMP2       16 INT "Temperature 2"
    POLY_READ_CONVERSION 0.0 0.01
    UNITS CELSIUS C
  APPEND_ITEM    MODE         8 UINT "Operating mode"
    STATE SAFE 0
    STATE NOMINAL 1
    STATE SCIENCE 2

TELEMETRY GMSECDEMO POWER BIG_ENDIAN "Power system, identified by APID"
  APPEND_ITEM    CCSDSVER     3 UINT "CCSDS version"
  APPEND_ITEM    CCSDSTYPE    1 UINT "CCSDS packet type"
  APPEND_ITEM    CCSDSSHF     1 UINT "CCSDS secondary header flag"
  APPEND_ID_ITEM CCSDSAPID   11 UINT 101 "CCSDS application process id"
  APPEND_ITEM    CCSDSSEQFLAGS 2 UINT "CCSDS sequence flags"
  APPEND_ITEM    CCSDSSEQCNT 14 UINT "CCSDS packet sequence count"
  APPEND_ITEM    CCSDSLENGTH 16 UINT "CCSDS packet data length"
  APPEND_ITEM    BUSVOLTAGE  16 UINT "Bus voltage"
    POLY_READ_CONVERSION 0.0 0.001
    UNITS VOLTS V
    LIMITS DEFAULT 1 ENABLED 24.0 26.0 32.0 34.0
  APPEND_ITEM    BUSCURRENT  16 UINT "Bus current"
    POLY_READ_CONVERSION 0.0 0.001
    UNITS AMPS A

TELEMETRY GMSECDEMO EVENT BIG_ENDIAN "Event text, identified by GMSEC subject"
  META GMSEC_SUBJECT GMSEC.DEMO.SAT1.MSG.TLM.EVENT
  APPEND_ITEM    CODE        16 UINT "Event code"
  APPEND_ITEM    TEXT         0 STRING "Event text"
