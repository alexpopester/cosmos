# Flow Automator: build automation flows by wiring nodes together.
#
# The tool is the graph editor. The FLOW_ENGINE microservice stores the flows,
# keeps their triggers armed and runs them server side, so flows keep running
# when nobody has the tool open.
#
# Variables:
#   flow_engine_port   port the engine's HTTP API listens on inside the operator
#   flow_route_prefix  Traefik route to the engine. Must be unique across scopes,
#                      and the tool expects /flow-automator.
#   trigger_period     seconds between trigger polls
#   run_history        runs kept per flow
#   contact_horizon_days  how far ahead contacts (passes) are predicted
#   retime_tolerance      seconds a contact may move after a TLE update and
#                         still carry its scheduled activities with it
#   stale_tle_days        TLEs older than this are flagged in the Contacts view
#   activity_retention_days  finished activities are deleted this long after
#                         their contact (run history and the audit log stay)

VARIABLE flow_engine_port 7790
VARIABLE flow_route_prefix /flow-automator
VARIABLE trigger_period 0.25
VARIABLE run_history 50
VARIABLE contact_horizon_days 7
VARIABLE retime_tolerance 120
VARIABLE stale_tle_days 3
VARIABLE activity_retention_days 7

MICROSERVICE FLOW_ENGINE flow-engine
  CMD python flow_engine.py
  OPTION TRIGGER_PERIOD <%= trigger_period %>
  OPTION HISTORY <%= run_history %>
  OPTION CONTACT_HORIZON_DAYS <%= contact_horizon_days %>
  OPTION CONTACT_RETIME_TOLERANCE <%= retime_tolerance %>
  OPTION STALE_TLE_DAYS <%= stale_tle_days %>
  OPTION ACTIVITY_RETENTION_DAYS <%= activity_retention_days %>
  PORT <%= flow_engine_port %>
  ROUTE_PREFIX <%= flow_route_prefix %>

TOOL flowautomator "Flow Automator"
  INLINE_URL main.js
  ICON mdi-sitemap-outline
